\begin{column}{0.44\textwidth}
\textbf{Open question after Phase I}:\\[0.3em]
\textit{Can adding a few TOP SiPMs on top of the END+Vik configuration improve resolution beyond what either readout alone achieves?}\\[0.6em]
\footnotesize
This requires new Geant4 simulation: \textbf{EndSparseTop} mode:
\begin{itemize}
  \item 16 END SiPMs (8 per face) + Vikuiti on $\pm Y$, $\pm Z$
  \item $N$ uniformly-spaced TOP SiPMs inside bar
  \item Vikuiti fills gaps between top SiPMs ($R{=}0.98$)
\end{itemize}
\vspace{0.3em}
Scans: EJ-230 + EJ-204, $N\in\{4,8,14,20\}$,\\
13 positions $\times$ 5000 ev — 104 ROOT files.
\end{column}

% ── 3. EndSparseTop geometry ──────────────────────────────────────────────────
\begin{frame}{EndSparseTop geometry}
\begin{columns}[T]
\begin{column}{0.54\textwidth}
\textbf{Volume layout (bar cross-section, $Y$--$Z$ view):}\\[0.5em]
\footnotesize
\begin{tabular}{ll}
\toprule
Face & Optical element\\
\midrule
$\pm X$ ends & 8 END SiPMs/face ($6\times6$ mm$^2$, 0.5 mm deep)\\
$+Y$ top & $N$ sparse TOP SiPMs, Vikuiti in gaps\\
$-Y$ bottom & Air gap + Vikuiti, $R{=}0.98$\\
$\pm Z$ sides & Air gap + Vikuiti, $R{=}0.98$\\
\bottomrule
\end{tabular}\\[0.6em]
\normalsize
\textbf{TOP SiPM placement}: uniformly spaced, step $= 1400/N$ mm.\\
Centre at $y = kBarHalfY - kTopHalfY\cdot 2 = 29.5$ mm\\
(0.25 mm inset below bar's $+Y$ face, so all SiPM faces\\
border bar material — avoids undefined AirGap$\to$SiPM optical boundary).\\[0.4em]
\textbf{Photon physics}:
\begin{itemize}\footnotesize
  \item Photon traveling $+Y$ hits SiPM at $y{=}29.25$ mm: absorbed/detected
  \item Photon missing a SiPM reaches bar face $y{=}30$ mm: Vikuiti reflects (98\%)
  \item Reflected photons get further chances at sparse SiPMs on each bounce
\end{itemize}
\end{column}
\begin{column}{0.42\textwidth}
\vspace{0.5em}
\textbf{Why top SiPMs outperform end SiPMs for timing:}\\[0.4em]
\footnotesize
\begin{tabular}{lcc}
\toprule
 & END SiPMs & TOP SiPMs\\
\midrule
Distance to track & $\sim$700 mm & $\sim$30 mm\\
Min.\ photon path & $\sim$700 mm & $\sim$30 mm\\
Path time (EJ-230) & $\sim$3.5 ns & $\sim$150 ps\\
Timing limited by & attenuation & scintillation $\tau$\\
\bottomrule
\end{tabular}\\[0.5em]
\normalsize
Top SiPMs capture near-field photons\\
with very short, nearly uniform paths.\\[0.3em]
With many photons and $k{=}1$ estimator:\\[0.2em]
$\sigma \approx \tau_\mathrm{rise}/\sqrt{N_\mathrm{pe,top}}$\\[0.2em]
\footnotesize
EJ-230 $\tau_\mathrm{rise}{=}0.5$ ns; $N_\mathrm{pe,top}{\sim}1000$ for $N{=}20$\\
$\Rightarrow \sigma \approx 16$ ps (matches simulation!)
\end{column}
\end{columns}
\end{frame}

\begin{column}{0.44\textwidth}
\textbf{Main results:}\\[0.3em]
\begin{enumerate}\footnotesize
  \item \textbf{EST $N{=}20$: 15 ps, position-independent}\\
        Top SiPMs with Vikuiti in gaps $\Rightarrow$ near-field timing\\
        ($\sigma \approx \tau_\mathrm{rise}/\sqrt{N_\mathrm{pe,top}} \approx 16$ ps theoretical).
  \item \textbf{Factor 3.3$\times$ gain over END+Vik alone}\\
        with 20 extra SiPMs over 1.4 m bar.
  \item \textbf{EST beats TOP-70 with 24 SiPMs} ($N{=}8$):
        28.9 ps vs 70.4 ps.
  \item \textbf{EJ-230 $\approx$ EJ-204} for EST: top SiPMs\\
        are fast enough that decay-time difference vanishes.
  \item \textbf{END contribution negligible} in BLUE for $N{\geq}8$:\\
        top SiPMs alone determine combined $\sigma$.
\end{enumerate}
\vspace{0.2em}
\normalsize
\textbf{Recommendation}: EJ-230 (or EJ-204) + EST $N{=}20$\\
as target geometry; $N{=}8$ if cost/coverage is critical.
\end{column}
