\begin{frame}{Rise time finito y CFD@14\%}
  \[
    I(t) \propto e^{-t/\tau_d}-e^{-t/\tau_r}, \qquad t>0
  \]
  \begin{itemize}
    \item Antes: emisión instantánea en $t_0$, con flanco temprano irreal.
    \item Ahora: perfil biexponencial con $(\tau_r,\tau_d)$ del datasheet.
    \item El flanco temprano fija el cruce CFD@14\%; por ello el rise time
      efectivo es crítico para una $\sigma_t$ defendible.
    \item Sin waveform, CFD@14\% se aproxima por el fotoelectrón
      $k=\mathrm{round}(0.14N_{\rm pe})$; no es un cruce sobre un pulso.
    \item End usa $(t_L+t_R)/2$ y exige al menos 10 PE en cada lado.
  \end{itemize}
\end{frame}

\begin{frame}{Definición de la resolución temporal}
  \small
  \begin{block}{Observable por evento: proxy CFD@14\%}
    Para cada grupo se ordenan los tiempos de todos los PE detectados y se toma
    $t_{14}=t_{(k)}$, con
    $k=\max[1,\mathrm{round}(0.14N_{\rm pe})]$.
    El 14\% se refiere al \textbf{total de PE detectados en ese grupo y evento},
    no al pico de un pulso reconstruido. Es un proxy de CFD; no modela todavía
    el CFD de FastIC+ ni el análisis de datos.
  \end{block}
  \begin{itemize}
    \item End: $t_{\rm End}=(t_{14,L}+t_{14,R})/2$, con
      $L=$ IDs 0--7 y $R=$ IDs 8--15; se exige $N_{\rm pe,L},N_{\rm pe,R}\geq10$
      en cada evento.
    \item Top: $t_{\rm Top}=t_{14}$ del conjunto agregado de arribos en IDs 16--35;
      se exige al menos un PE.
    \item $N_{\rm eff}$ es el número de eventos que sobrevive estos cortes.
  \end{itemize}
\end{frame}

\begin{frame}{Definición operativa de $\sigma_t$ y alcance}
  \small
  \begin{itemize}
    \item Para cada distribución de $t_{\rm End}$ o $t_{\rm Top}$ se ajusta el
      \textbf{core gaussiano}: histograma de 100 bins centrado en la mediana,
      rango $\pm8\sigma_{\rm robusta}$, con
      $\sigma_{\rm robusta}=1.4826\,\mathrm{MAD}$, y hasta 4 ajustes iterativos
      en $\pm2\sigma$.
    \item Se cita $\sigma_t=\sigma_{\rm fit}\pm\delta\sigma_{\rm fit}$ y
      $\chi^2/\mathrm{ndf}$. RMS y FPT (primer PE) son controles; si el fit falla,
      el código cae a RMS y marca $\chi^2/\mathrm{ndf}=-1$.
    \item Las curvas plegadas concatenan eventos de $+|x|$ y $-|x|$ y
      \textbf{vuelven a ajustar}. End se considera fiable cuando el yield medio
      de ambos lados es $\geq10$ PE: en este scan, $|x|\leq200$ mm.
    \item Run: $\mu^-$ 1 GeV, EJ-204, rise time finito, 10\,000 eventos/posición,
      jitter de sensor/electrónica = 0.
  \end{itemize}
  \begin{alertblock}{Caveat duro}
    $\sigma_t$ mostrado es \textbf{intrínseco}: centelleo biexponencial,
    dispersión óptica, estadística de fotones y estimador.
    $\sigma_{\rm total}=\sqrt{\sigma_{\rm intr}^2+\sigma_{\rm SPTR}^2+
    \sigma_{\rm elec}^2}$; aquí solo se muestra $\sigma_{\rm intr}$, aún no
    comparable con los 250 ps del test beam.
  \end{alertblock}
\end{frame}

\begin{frame}{Validación de simetría tras alta estadística}
  \begin{columns}[T,onlytextwidth]
    \column{0.47\textwidth}
    \centering
    \includegraphics[width=\textwidth]{\results/sigma_cfd_unfolded_vs_position.pdf}
    \column{0.50\textwidth}
    \scriptsize
    \begin{block}{Métrica}
      \[
        A_\sigma(|x|)=
        \frac{2[\sigma(+x)-\sigma(-x)]}
             {\sigma(+x)+\sigma(-x)}
      \]
      Se evalúa sin plegar; las curvas finales de $\sigma$ combinan $\pm|x|$.
    \end{block}
    \begin{itemize}
      \item Yield End-L/End-R simétrico: geometría consistente.
      \item La asimetría residual de $\sigma$ cae frente al scan de 200 eventos.
      \item Cualquier residuo significativo persistente apunta al análisis,
        no a la física simétrica.
    \end{itemize}
    \begin{block}{Resultado}
      \small Para $\sigma_{\rm fit}$ intrínseco del proxy CFD@14\%:
      End fiable, mediana $|A_\sigma|=2.35\%$, máximo $4.15\%$;
      Top, mediana $|A_\sigma|=1.12\%$. La asimetría preliminar estaba
      dominada por estadística.
    \end{block}
  \end{columns}
\end{frame}

\begin{frame}{Lectura física}
  \begin{itemize}
    \item \textbf{Top readout}: SiPMs a milímetros de la traza capturan fotones
      antes de acumular dispersión por propagación; yield relativamente estable
      y $\sigma_{\rm Top}<\sigma_{\rm End}$ a nivel intrínseco.
    \item \textbf{End readout}: atenuación y dispersión de caminos degradan el
      lado lejano; cuando queda hambriento, $(t_L+t_R)/2$ deja de ser fiable.
    \item \textbf{Trade-off EJ-204}: atenuación de 160 cm frente a 380 cm de
      EJ-200 produce yields End más bajos y empinados, a cambio de centelleo
      más rápido ($0.7/1.8$ ns).
  \end{itemize}
  \begin{block}{Titular defendible}
    Top mejora $\sigma_{\rm fit}$ intrínseco del proxy CFD@14\% frente a End
    por un factor mediano de 2.54 para $|x|\leq200$ mm. Los valores absolutos
    quedan pendientes del jitter real y de aplicar un CFD común a sim y datos.
  \end{block}
\end{frame}
