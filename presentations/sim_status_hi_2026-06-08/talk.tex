\begin{frame}{Definición de timing congruente con el test beam}
  \small
  \begin{block}{End: SUM4 + self-trigger}
    Se forman clusters provisionales $\{0..3\}$, $\{4..7\}$,
    $\{8..11\}$ y $\{12..15\}$. Para cada cluster se suman respuestas 1-PE
    desplazadas por los tiempos de arribo y se toma el primer cruce
    \textbf{leading-edge}. En cada extremo, $t_L$ o $t_R$ es el primer cluster
    que cruza el umbral.
  \end{block}
  \[
    \Delta T_{LR}=t_L-t_R,\qquad
    \sigma_{\rm single,End}=\frac{\sigma_{\rm core}(\Delta T_{LR})}{\sqrt{2}}.
  \]
  \begin{itemize}
    \item $\sigma_{\rm core}$: fit gaussiano iterativo en $\pm2\sigma$;
      se reporta $\sigma\pm\delta\sigma$, $\chi^2/\mathrm{ndf}$ y $N_{\rm eff}$.
    \item Constanza observa leading-edge + walk; en etapa 1 no hay corrección
      de walk ni corte ToT.
  \end{itemize}
\end{frame}

\begin{frame}{Caveat duro: etapa 1 intrínseca}
  \[
    \sigma_{\rm total} =
    \sqrt{\sigma_{\rm intr}^2+\sigma_{\rm SPTR}^2+\sigma_{\rm elec}^2}
  \]
  \begin{itemize}
    \item Aquí $\sigma_{\rm intr}$ incluye centelleo biexponencial, dispersión
      óptica, estadística de fotones, SPR provisional, leading-edge y selección.
    \item No incluye SPTR ni jitter electrónico; tampoco walk corregido ni
      cortes ToT del dato.
    \item El blanco de 88 ps se muestra solo como referencia cualitativa.
      No se afirma igualdad ni comparabilidad absoluta en esta etapa.
    \item El LSB temporal indicado por las slides de Constanza es
      aproximadamente \textbf{24 ps}; debe confirmarse desde su repo.
  \end{itemize}
\end{frame}
