\begin{frame}{Datasets and analysis strategy}
\begin{columns}[T]
\begin{column}{0.48\textwidth}
\textbf{Available ROOT files} (all 12 branches)\\
\footnotesize
\begin{tabular}{lrcc}
\toprule
Dataset & Files & Readout & ph/ev/end\\

\bottomrule
\end{tabular}\\[0.4em]
\normalsize
Key branches: \texttt{event\_id}, \texttt{face\_type}, \texttt{global\_id},\\
\texttt{local\_id}, \texttt{time\_ns}, \texttt{gun\_x\_mm}\\[0.4em]
\textbf{What is computable offline:}
\begin{itemize}\footnotesize
  \item $k$-th order statistic: any $k \le N_\mathrm{pe}$
  \item Mean of first $m$ photons per end
  \item Software SiPM ablation via \texttt{local\_id} mask
  \item MC-truth bias via \texttt{gun\_x\_mm}
\end{itemize}
\end{column}
\begin{column}{0.48\textwidth}
\textbf{Analysis phases:}\\[0.3em]
\begin{description}\footnotesize
  \item[Phase A] $k$-scan: $T_0^{(k)}=\tfrac{1}{2}[t_L^{(k)}+t_R^{(k)}]$, $k=1\ldots200$
  \item[Phase B] $m$-average: $T_0^{(m)}=\tfrac{1}{2}[\bar{t}_L^{(1:m)}+\bar{t}_R^{(1:m)}]$
  \item[Phase C] TOP ablation: $N_\mathrm{top}$ from 70 to 1 (8 random seeds)
  \item[Phase D] END ablation: $N_\mathrm{end}$ from 8 to 1 per face (8 random seeds)
  \item[Phase E] Pareto frontier: $\sigma(T_0)$ vs SiPM count
\end{description}
\vspace{0.5em}
\textbf{No new Geant4 runs required for these.}\\[0.3em]
All results use IQR-based $\sigma$ (robust to non-Gaussian tails).\\[0.2em]
EJ-230: 13 positions $\times$ 5000 ev; $N_\mathrm{pe}/\mathrm{end}$: 350--1540.
\end{column}
\end{columns}
\end{frame}

% ── 8. What requires new Geant4 ───────────────────────────────────────────────
\begin{frame}{What genuinely requires new Geant4 simulation}
\begin{columns}[T]
\begin{column}{0.50\textwidth}
\textbf{Computable from existing data:}\\[0.3em]
\begin{itemize}\footnotesize
  \item $k$-scan and $m$-average, any $k,m \le N_\mathrm{pe}$
  \item END software ablation (8 to 1 SiPM/face)
  \item TOP software ablation (70 to 1 SiPMs)
  \item Position dependence of all above
  \item MC-truth bias $\langle T_0 - T_0^\mathrm{true}\rangle$ vs.\ $x$
\end{itemize}
\vspace{0.5em}
\textbf{NOT computable from existing data:}\\[0.3em]
\begin{enumerate}\footnotesize
  \item \textbf{END+Vik + sparse TOP simultaneously}\\
        In \texttt{scan\_resolution}: top SiPMs absorb photons $\Rightarrow$ end hits $\approx$1/ev.\\
        Need combined geometry: Vikuiti at ends + $\mathcal{O}$(few) top SiPMs.
  \item \textbf{Physical SiPM reduction} ($< 8$/end):\\
        Software masking changes information, not optical boundaries.\\
        True optical effect of removing SiPMs from face requires new sim.
  \item \textbf{EJ-200 position scan with reflector}:\\
        Only \texttt{scan\_end\_vikuiti} has EJ-200; additional systematic checks possible.
\end{enumerate}
\end{column}
\begin{column}{0.46\textwidth}
\textbf{Proposed minimal new campaign:}\\[0.5em]
\footnotesize
\begin{tabular}{lll}
\toprule
Geometry & Scint. & Goal\\
\midrule
END+Vik + 4 TOP & EJ-230 & Combined $T_0$\\
END+Vik + 8 TOP & EJ-230 & Combined $T_0$\\
4 SiPMs/end (phys.) & EJ-230 & True ablation\\
2 SiPMs/end (phys.) & EJ-230 & Minimum config\\
\bottomrule
\end{tabular}\\[0.4em]
\normalsize
$\approx 4$ new geometries $\times$ 13 positions $\times$ 5000 ev\\
$\approx$ 4 hours total runtime on t0minidaq.\\[0.3em]
Can reuse run-script infrastructure.
\end{column}
\end{columns}
\end{frame}

\footnotesize

\toprule
\textbf{Configuration} & \textbf{SiPMs} & $\boldsymbol{\sigma(T_0)}$\\
\midrule

EJ-230 + END + Vik, 2/end, $k^*$   & 4  & 97.9 ps\\
\bottomrule
\end{tabular}
\end{column}
\begin{column}{0.44\textwidth}
\textbf{Main results:}\\[0.3em]
\begin{enumerate}\footnotesize
  \item \textbf{END+Vikuiti beats TOP at equal SiPM cost} by a factor of $\approx$2--5.
  \item \textbf{$m$-average $\sim$10\% better} than single $k$-th photon; optimal $m^*=5$--12.
  \item \textbf{EJ-230 best at centre}; EJ-200 competes at extremes (longer $L_\mathrm{att}$).
  \item \textbf{4 END SiPMs total $\to$ $<100$ ps} (software ablation).
  \item \textbf{TOP needs $\gtrsim50$ SiPMs} for comparable timing.
  \item \textbf{Combination} END+Vik + sparse TOP requires new simulation.
\end{enumerate}
\vspace{0.3em}
\normalsize
\textbf{Recommendation}: run EJ-230 + END+Vik with $m^*=7$ as baseline; explore physical 4-SiPM/end geometry with new Geant4 campaign.
\end{column}
\end{columns}
