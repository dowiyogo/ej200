\begin{frame}{Modelo \'optico: TIR vs.\ Vikuiti}
  \begin{columns}[T]
    \column{0.50\textwidth}
    \textbf{TIR only}
    \begin{itemize}
      \item Interfaz barra--aire: \texttt{dielectric\_dielectric}, pulida.
      \item TIR para $\theta > \theta_c = 39.3^\circ$ ($n=1.58$).
      \item Fotones que no superan $\theta_c$ \alert{escapan} por la cara.
      \item Sin paneles reflectores: cara $+Y$ libre (o con SiPMs).
    \end{itemize}

    \vspace{0.5em}
    \textbf{Vikuiti 3M ESR (air gap)}
    \begin{itemize}
      \item Air gap \SI{0.10}{mm} entre barra y panel Vikuiti.
      \item TIR preservada en interfaz barra--aire.
      \item Fotones que escapan la TIR son reflejados: $R = 0.98$.
      \item Paneles en las 4 caras laterales ($\pm Y$, $\pm Z$).
      \item \textbf{Solo los extremos $\pm X$ quedan abiertos.}
    \end{itemize}

    \column{0.47\textwidth}
    \begin{alertblock}{Vikuiti skin (sin air gap)}
      Modelo alternativo: \texttt{dielectric\_metal} directamente
      sobre la barra. \textbf{Elimina la TIR.} Estudiado para TOP readout.
      Resultado: id\'entico a TIR (SiPMs top ya capturan todo).
    \end{alertblock}
    \vspace{0.4em}
    \begin{block}{GROUPVEL fix}
      Aire del mundo: $v_g = c/1.58$ para evitar
      velocidades de grupo supralum\'inicas en Geant4.
    \end{block}
  \end{columns}
\end{frame}

      \toprule
      Config & $N_{\rm pe}$ & $\sigma(T_0)$ \\
      \midrule
      EJ-204 END+TIR & $\sim$1 & $\sim$1200\,ps \\
      EJ-230 END+TIR & $\sim$1 & $\sim$970\,ps  \\[2pt]
      EJ-204 END+Vik & 1882    & 63\,ps \\
      EJ-230 END+Vik & 1403    & 59\,ps \\
      \bottomrule
    \end{tabular}

    \vspace{0.5em}
    \textbf{Por qu\'e END+TIR falla:}
    \begin{itemize}
      \item Sin reflector, los fotones que no superan $\theta_c$ escapan por las 4 caras.
      \item Solo $\sim$1 fot\'on/evento llega a cada extremo.
      \item Timing dom. por estad\'istica de Poisson, no por la f\'isica del centelleador.
    \end{itemize}

    \textbf{Con Vikuiti:}
    \begin{itemize}
      \item $\times$\,1000 m\'as fotones en los extremos.
      \item $\sigma(T_0)$ compite con TOP readout.
    \end{itemize}
    }
  \end{columns}
\end{frame}

      \item EJ-204: $\sigma(\Delta t)\approx\SI{120}{ps}$ $\Rightarrow$ $\sigma_x\approx\SI{11.0}{mm}$

\begin{frame}{Conclusiones}
  \begin{columns}[T]
    \column{0.56\textwidth}
    \textbf{Hallazgos principales}
    \begin{enumerate}
      \item \textbf{Vikuiti no mejora el TOP readout.}
            Los SiPMs superiores ya capturan todos los fotones incidentes;
            el reflector no puede agregar m\'as.

      \item \textbf{END+TIR es inviable.}
            Sin reflector, los fotones escapan por las 4 caras laterales:
            $N_{\rm pe}\sim 1$, $\sigma(T_0)\sim 1$\,ns.

      \item \textbf{END+Vikuiti (air gap) es la mejor configuraci\'on.}
            Combina TIR nativa con reflexi\'on Vikuiti ($R=0.98$) en las caras laterales:
            todos los fotones se guian hacia los extremos.

      \item \textbf{EJ-230 es el mejor centelleador} para END+Vikuiti:
            $\sigma(T_0) = \mathbf{59}$\,\textbf{ps}, mejora $\sim$\textbf{19\%}
            sobre el mejor TOP readout (73\,ps con EJ-230+TIR).

      \item \textbf{Resoluci\'on de posici\'on}: $\sigma_x = \SI{10.3}{mm}$
            (EJ-230, END+Vikuiti), con $v_{\rm eff} = \SI{184.6}{mm/ns}$.
    \end{enumerate}

    \column{0.42\textwidth}
    \begin{block}{\textcolor{bestgreen}{Configuraci\'on \'optima}}
      \centering
      \textbf{EJ-230 + END + Vikuiti air-gap}\\[0.5em]
      \begin{tabular}{ll}
        $\sigma(T_0)$     & \textbf{59 ps} \\
        $\sigma_x$        & \textbf{10.3 mm} \\
        $N_{\rm pe}$      & 1403 (en $x=0$) \\
        $v_{\rm eff}$     & 184.6 mm/ns \\
        $n_{\rm eff}$     & 1.62 \\
      \end{tabular}
    \end{block}
    \vspace{0.3em}
    \begin{alertblock}{Pr\'oximos pasos}
      \begin{itemize}
        \item Jitter electr\'onico: $\sigma_j \approx \SI{100}{ps}$
              $\Rightarrow$ $\sigma_{\rm total} = \sqrt{59^2 + 100^2} \approx 117$\,ps.
        \item Combinar END + TOP para mejorar a\'un m\'as.
        \item Optimizar $k$-\'esimo fot\'on en END readout.
      \end{itemize}
    \end{alertblock}
  \end{columns}
\end{frame}
