\begin{frame}{Motivación del experimento}
  \begin{columns}[T]
    \column{0.55\textwidth}
    \textbf{Objetivo}
    \begin{itemize}
      \item Estudiar la resolución temporal y de posición de la barra
            usando \textit{únicamente} los SiPM de extremo (End).
      \item Eliminar la lectura superior (Top) para aislar el comportamiento
            longitudinal de la barra.
      \item Comparar tres centelleadores con distintas propiedades cinéticas.
    \end{itemize}
    \vspace{0.4em}
    \textbf{Reflexión óptica}
    \begin{itemize}
      \item Vikuiti 3M ESR: $R = 0.98$, superficie pulida.
      \item \alert{Air gap de \SI{0.10}{mm}} entre barra y reflector.
      \item La interfaz barra--aire genera reflexión interna total (TIR)
            para ángulos $\theta > \theta_c = 39.3°$ ($n_{\rm bar} = 1.58$).
      \item El reflector Vikuiti captura los fotones que escapan la TIR.
    \end{itemize}

    \column{0.43\textwidth}
    \begin{block}{Geometría de la barra}
      \centering
      $1400 \times 60 \times 10\ \text{mm}^3$\\[0.4em]
      \begin{tabular}{ll}
        End-left  & 8 SiPM (cara $-X$)\\
        End-right & 8 SiPM (cara $+X$)\\
        Top       & \textbf{sin leer}   \\[0.2em]
        Muón      & $\mu^-$, \SI{1}{GeV}, vertical\\
        Jitter    & \SI{0}{ns}\\
        Eventos   & 5000 por posición\\
        Posiciones & 13 ($-600$ a $+600$ mm)\\
      \end{tabular}
    \end{block}
  \end{columns}
\end{frame}

\begin{frame}{Modelo óptico: Vikuiti con Air Gap}
  \begin{columns}[T]
    \column{0.55\textwidth}
    \textbf{Implementación en Geant4}
    \begin{itemize}
      \item Cuatro paneles explícitos alrededor de la barra:
            $-Y$, $+Y$, $+Z$, $-Z$.
      \item Cada panel: capa de \textbf{aire} (\SI{0.10}{mm}) +
            capa de \textbf{reflector} (\SI{0.05}{mm}).
      \item La interfaz aire--reflector usa \texttt{G4LogicalBorderSurface}\\
            con $R(\lambda) = 0.98$, \texttt{dielectric\_metal}.
      \item La interfaz barra--aire mantiene \texttt{dielectric\_dielectric}
            (TIR activada naturalmente).
    \end{itemize}
    \vspace{0.4em}
    \textbf{Comparación con Vikuiti skin surface}
    \begin{itemize}
      \item El modelo de ``skin surface'' (\texttt{dielectric\_metal}
            directamente sobre la barra) \textbf{elimina la TIR}.
      \item El modelo air-gap preserva la TIR \textit{y} recupera los fotones
            que la escapan: física correcta.
    \end{itemize}

    \column{0.43\textwidth}
    \begin{block}{GROUPVEL fix}
      El mundo (aire) recibe:
      \begin{equation*}
        v_g = \frac{c}{n_{\rm bar}} = \frac{c}{1.58}
      \end{equation*}
      para evitar velocidades de grupo supralumínicas que corrompían
      los tiempos de llegada en Geant4.
    \end{block}
    \vspace{0.3em}
    \begin{alertblock}{Validación previa}
      100 eventos: EJ-200 $\to$ 2651 ph/ev,
      EJ-204 $\to$ 1896 ph/ev, EJ-230 $\to$ 1361 ph/ev.\\
      Sin SiPMs top (face\_type=2: 0 hits). ✓
    \end{alertblock}
  \end{columns}
\end{frame}

\begin{frame}{Definición de las observables}
  \begin{columns}[T]
    \column{0.55\textwidth}
    \textbf{Tiempo de llegada} al extremo $L$ ($R$):
    \begin{equation*}
      t_{L,R}^{(k)} = k\text{-ésimo fotón más temprano en cara }-X\ (+X)
    \end{equation*}
    Con $k=1$ (primer fotón). Alta sensibilidad al tiempo de tránsito óptico.

    \vspace{0.6em}
    \textbf{Tiempo de vertice (T0):}
    \begin{equation*}
      T_0 = \frac{t_L^{(1)} + t_R^{(1)}}{2}
      \quad \Rightarrow \quad \sigma(T_0):\ \text{resolución temporal}
    \end{equation*}
    $T_0$ es independiente de la posición $x$ a primer orden.

    \vspace{0.4em}
    \textbf{Diferencia de tiempos:}
    \begin{equation*}
      \Delta t = t_R^{(1)} - t_L^{(1)}
      = \frac{-2x}{v_{\rm eff}} + \text{const}
    \end{equation*}
    $\sigma(\Delta t) \to$ resolución de posición;
    slope $\to v_{\rm eff}$.

    \column{0.43\textwidth}
    \begin{block}{Análisis estadístico}
      Por posición y centelleador:
      \begin{enumerate}
        \item Colectar $t_L^{(1)}, t_R^{(1)}$ por evento.
        \item Ajuste gaussiano via IQR sobre distribuciones de
              $T_0$ y $\Delta t$.
        \item Ajuste lineal de $\langle\Delta t\rangle$ vs.\ $x$
              para extraer $v_{\rm eff}$.
      \end{enumerate}
    \end{block}
    \vspace{0.3em}
    \begin{alertblock}{Nota}
      Sin jitter electrónico ($\sigma_j = \SI{0}{ns}$):
      los resultados son un límite inferior del hardware real.
    \end{alertblock}
  \end{columns}
\end{frame}

\begin{frame}{Resolución temporal $\sigma(T_0)$ vs.\ posición}
  \begin{columns}[T]
    \column{0.64\textwidth}
    \includegraphics[width=\textwidth]{\plots/end_vik_sigma_T0_vs_x.pdf}
    \column{0.34\textwidth}
    \begin{itemize}
      \item $T_0 = (t_L + t_R)/2$ es prácticamente plano en $x$
            (variación $<5$ ps en todo el rango).
      \item \textbf{EJ-230 gana}: $\sigma(T_0)\approx57$--$63$ ps
            gracias a $\tau_d = \SI{1.5}{ns}$.
      \item \textbf{EJ-204}: $\sigma(T_0)\approx61$--$65$ ps;
            compensa menor yield con $\tau_d$ más corto que EJ-200.
      \item \textbf{EJ-200}: $\sigma(T_0)\approx63$--$68$ ps,
            penalizado por $\tau_d = \SI{2.1}{ns}$ pese a mayor yield.
      \item Mejora EJ-230 vs EJ-200: \textbf{$\sim$10\%} en $\sigma(T_0)$.
    \end{itemize}
  \end{columns}
\end{frame}

% ─────────────────────────────────────────────────────────────────────────────
\begin{frame}{Resolución de posición $\sigma(\Delta t)$ vs.\ posición}
  \begin{columns}[T]
    \column{0.64\textwidth}
    \includegraphics[width=\textwidth]{\plots/end_vik_sigma_dt_vs_x.pdf}
    \column{0.34\textwidth}
    \begin{itemize}
      \item $\Delta t = t_R - t_L$ mide la posición longitudinal $x$.
      \item $\sigma(\Delta t) \to$ resolución de posición vía:
            \begin{equation*}
              \sigma_x = \frac{v_{\rm eff}}{2}\,\sigma(\Delta t)
            \end{equation*}
      \item $\sigma(\Delta t)$ es bastante plano en $x$.
            Ligero aumento hacia los extremos en EJ-204/EJ-230
            por atenuación.
      \item \textbf{EJ-230}: $\sigma(\Delta t)\approx108$--$120$ ps.
      \item \textbf{EJ-204}: $\sigma(\Delta t)\approx116$--$125$ ps.
      \item \textbf{EJ-200}: $\sigma(\Delta t)\approx118$--$127$ ps.
      \item Resolución de posición (ver tabla): $\sigma_x \sim 10$--$12$ mm.
    \end{itemize}
  \end{columns}
\end{frame}

% ─────────────────────────────────────────────────────────────────────────────
\begin{frame}{Velocidad de grupo efectiva $v_{\rm eff}$}
  \begin{columns}[T]
    \column{0.64\textwidth}
    \includegraphics[width=\textwidth]{\plots/end_vik_groupvel.pdf}
    \column{0.34\textwidth}
    {\footnotesize
    \begin{itemize}
      \item Relación lineal: $\langle\Delta t\rangle = -2x/v_{\rm eff} + \text{cte}$.
      \item $v_{\rm eff} < c/n_{\rm bar} = \SI{190}{mm/ns}$
            porque los fotones zigzaguean por TIR:
            \begin{equation*}
              v_x = v_g\,\sin\theta < v_g
            \end{equation*}
      \item \textbf{Resultado}: $v_{\rm eff} \approx \SI{184.6}{mm/ns}$,
            $n_{\rm eff} \approx 1.62$.
      \item Idéntico para los tres: propiedad geométrica. ✓
      \item \textbf{Contraste con Top readout}:
            $v_{\rm eff}^{\rm top} \approx \SI{145}{mm/ns}$,
            $n_{\rm eff}^{\rm top} \approx 2.07$.
      \item Los primeros fotones en End viajan casi paralelos
            al eje $X$; los del Top viajan más diagonalmente.
    \end{itemize}
    }
  \end{columns}
\end{frame}

\begin{frame}{Conclusiones}
  \begin{columns}[T]
    \column{0.55\textwidth}
    \textbf{Hallazgos principales}
    \begin{itemize}
      \item \textbf{EJ-230 gana en resolución temporal:}
            $\sigma(T_0) \approx 58.6$ ps vs.\ 63.2 (EJ-204) vs.\ 65.2 ps (EJ-200)
            en $x=0$. Mejora $\sim$10\% sobre EJ-200.
      \item \textbf{EJ-200 gana en yield:}
            2621 ph/ev vs.\ 1881 vs.\ 1402 en $x=0$;
            más uniforme (menor atenuación).
      \item $v_{\rm eff} = 184.6$ mm/ns idéntico para los tres:
            propiedad geométrica de TIR. ✓
      \item Contraste con Top readout ($v_{\rm eff}^{\rm top}=145$ mm/ns):
            los primeros fotones en End viajan más horizontalmente.
      \item $\sigma(T_0)$ plano en $x$ ($<5$ ps de variación): $T_0$
            funciona como estimador de tiempo de vertice. ✓
    \end{itemize}
    \vspace{0.3em}
    \textbf{Próximos pasos}
    \begin{itemize}
      \item Añadir jitter electrónico ($\sigma_j \approx \SI{100}{ps}$).
      \item Comparar End-only vs.\ Top+End combinado.
