\section{Timing Definitions}

% ---- Frame 6 -------------------------------------------------------
\begin{frame}{Timing estimators — order statistics}
\begin{block}{Order-statistic definition}
For a single channel and single event, sort the detected hits:
\[
  t^{(1)} \le t^{(2)} \le \cdots \le t^{(k)} \le \cdots
\]
\kfour{}: $t^{(4)}$ is the 4th smallest arrival time.\\
\ktwenty{}: $t^{(20)}$ is the 20th smallest arrival time.
\end{block}
\vspace{8pt}
\begin{columns}
  \column{0.48\textwidth}
  \begin{alertblock}{Differential observable (position)}
    \[
      \Delta t^{(k)} = t_A^{(k)} - t_B^{(k)}
    \]
    Calibrated as: $\mu_{\Delta t}(x) = m_k x + b_k$
  \end{alertblock}
  \column{0.48\textwidth}
  \begin{exampleblock}{Common observable (event time)}
    \[
      t_+^{(k)} = \frac{t_A^{(k)} + t_B^{(k)}}{2}
    \]
    Sensitive to absolute arrival time of the particle.
  \end{exampleblock}
\end{columns}
\end{frame}

% ---- Frame 22 -------------------------------------------------------
\begin{frame}{Does 20PE really improve performance?}
\begin{columns}
  \column{0.50\textwidth}
    \begin{exampleblock}{Where 20PE helps}
      \begin{itemize}
        \item Steeper calibration slope
              ($\SlopeTwentyPE$ vs.\ $\SlopeFourPE$\,ps/mm)
        \item Better mean position resolution
              (\RmsxTwentyPE{} vs.\ \RmsxFourPE\,mm)
        \item More photons averaged $\Rightarrow$
              more stable estimator in some regimes
      \end{itemize}
    \end{exampleblock}
  \column{0.48\textwidth}
    \begin{alertblock}{Where 20PE does not help}
      \begin{itemize}
        \item Wider timing spread
              ($\SigmaDtTwentyPE$ vs.\ \SigmaDtFourPE\,ps)
        \item Worse calibration linearity
              ($\chi^2/\mathrm{ndf} = \ChiTwentyPE$)
        \item Larger A–B correlation
              ($\rho = \RhoABTwentyPE$ vs.\ \RhoABFourPE)
        \item Larger maximum bias
              (\BiasMaxTwentyPE{} vs.\ \BiasMaxFourPE\,mm)
      \end{itemize}
    \end{alertblock}
\end{columns}
\vspace{4pt}
\centering\small
\textbf{Verdict:} 20PE improves position resolution via slope,
not via timing precision. The improvement comes with a cost in
linearity and common-mode contamination.
\end{frame}

\section{Conclusions and Next Steps}

% ---- Frame 28 -------------------------------------------------------
\begin{frame}{Conclusions}
\begin{enumerate}
  \item \textbf{Timing:}
        \FourPElabel{} gives $\sigma_{\Delta t}\approx\SigmaDtFourPE$\,ps;
        \TwentyPElabel{} gives $\approx\SigmaDtTwentyPE$\,ps (27\% wider).
  \item \textbf{Position:}
        20PE yields better mean LOO-CV resolution
        (\RmsxTwentyPE{} vs.\ \RmsxFourPE\,mm)
        due to steeper calibration slope
        ($\SlopeTwentyPE$ vs.\ $\SlopeFourPE$\,ps/mm).
  \item \textbf{A–B correlation:}
        rises from $\rho=\RhoABFourPE$ (4PE) to $\rho=\RhoABTwentyPE$ (20PE),
        indicating increased common-mode contamination.
  \item \textbf{Calibration:}
        20PE $\chi^2/\mathrm{ndf}=\ChiTwentyPE$ signals non-linearity;
        4PE fits are more linear ($\chi^2=\ChiFourPE$).
  \item \textbf{Efficiency:}
        100\% for both thresholds in the fine scan — no efficiency penalty.
  \item \textbf{Conceptual:}
        $t_{20}$ (20th detected hit) $\ne$ 20\% CFD of Lv et al.
  \item \textbf{Scope:}
        All numbers are \highlight{intrinsic simulation predictions}
        (jitter = 0; no SPTR; no electronics).
\end{enumerate}
\end{frame}

\begin{frame}{B3 — Fit QA criteria}
\small
Gaussian fit procedure applied to $\Delta t$ distributions at each position:
\begin{enumerate}
  \item Histogram with documented binning
  \item Seed: mean at bin-maximum
  \item Robust seed for $\sigma$: 1.48$\times$MAD
  \item Fit window: $\mu \pm 2.5\,\sigma_\mathrm{seed}$
  \item Iterate: refit in $\mu_\mathrm{fit} \pm 2.5\,\sigma_\mathrm{fit}$
  \item Record: status, covariance, $\chi^2$/ndf, fraction in $\pm3\sigma$,
        range used
\end{enumerate}
\vspace{4pt}
If $n < 50$ or distribution is multi-modal: report median, RMS$_{68}$,
MAD~$\sigma$ instead. No fit forced on degenerate distributions.
\end{frame}

% ---- B6 -------------------------------------------------------
\begin{frame}{B6 — LOO cross-validation methodology}
\small
\textbf{Procedure:}
\begin{enumerate}
  \item For each of the 41 positions, exclude it from the calibration set
  \item Fit $\mu_{\Delta t}(x) = m x + b$ on remaining 40 positions
  \item Reconstruct $x_\mathrm{rec} = (\Delta t - b) / m$ event-by-event
        at the excluded position
  \item Collect all 3000 events; compute bias, RMS$_{68}$, $\sigma_\mathrm{core}$
  \item Repeat for all 41 positions
\end{enumerate}
\vspace{4pt}
\textbf{Stored outputs:}
\begin{itemize}
  \item \texttt{loo\_predictions\_4pe.npz}: 123\,000 events with
        \texttt{x\_true} and \texttt{x\_rec}
  \item \texttt{loo\_predictions\_20pe.npz}: same structure
  \item \texttt{position\_reconstruction\_\{4,20\}pe.csv}:
        per-position summary (bias, RMS$_{68}$, valid fraction)
\end{itemize}
\end{frame}
